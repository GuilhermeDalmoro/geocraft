///@function cria_item
function cria_item(_meu_id, _nome, _desc, _spr, _frame,  _dano, _cura, _atributo, _rar, _vel, _tipo) constructor {
    static qtd_item = 0;
    item_id = qtd_item++;
	
	meu_id = _meu_id;
	
    nome = _nome;
    desc = _desc;
    spr = _spr;
	frame = _frame
    dano = _dano;
    cura = _cura;
    atributo = _atributo;
    rar = _rar;
    vel = _vel;
	tipo = _tipo;
    
    usa_item = function() {
        global.item_player = global.itens[| item_id];
    }
    
    coleta_item = function() {
        var _colunas = ds_grid_width(global.inventario_slots);
        var _linhas = ds_grid_height(global.inventario_slots);
        for (var _i = 0; _i < _linhas; _i++) {
            for (var _j = 0; _j < _colunas; _j++) {
                var _atual = global.inventario_slots[# _j, _i];
                if (!_atual) {
                    global.inventario_slots[# _j, _i] = global.itens[| item_id];
                    return true;
                }
            }
        }
        return false;
    }
}

dropa_item = function(){
	var _ind = 0;
	var _chance = random(100)
	
	switch(dropavel){
		case "pedra":
		 _ind = 0;
		break;
		
		case "madeira":
		_ind = 1;
		break;
		
		case "cerca":
		_ind = 2;
		break;
		
		case "flor":
		_ind = 3;
		break;
		
		case "bloco_pedra":
		_ind = 4;
		break;
		
		case "blmadeira":
		_ind = 5;
		break;
		
		case "bloco_terra":
		_ind = 6;
		break;
		
		}
		
		//criando o dropavel
		var _repete = 1;
		
		if (_chance >= 99){
			_repete = 8
		}else if (_chance > 60 ){
			_repete = 4
		}else if (_chance > 50 ){
			_repete = 2
		}else if (_chance > 30 ){
			_repete = 1
		}
		
		
		repeat(_repete){
			var _x = x + random_range(-10, 10);
			var _y = y+ random_range(-10, 10)
			var _drop = instance_create_depth(_x, _y, depth, obj_drop);
			_drop.image_index = _ind;
		
			instance_destroy();
		}
}
